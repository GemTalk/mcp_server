# Demo B

curl -i -X POST localhost:8000/mcp -H 'Content-Type: application/json' -d '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-11-25","capabilities":{},"clientInfo":{"name":"demo","version":"1.0"}}}'

## Copy the MCP-Session-Id off the header and park it (32 hex chars, no quoting needed):
SID=<paste>

curl -s -X POST localhost:8000/mcp -H 'Content-Type: application/json' -H "MCP-Session-Id: $SID" -d '{"jsonrpc":"2.0","id":2,"method":"tools/list"}'

## If the room asks to see the session end -- then the cache line again, the row is gone:
curl -i -X DELETE localhost:8000/mcp -H "MCP-Session-Id: $SID"



# Demo C

## 8010 has its own sessions: $SID is not valid here. Initialize against it first.
curl -i -X POST localhost:8010/mcp -H 'Content-Type: application/json' -d '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-11-25","capabilities":{},"clientInfo":{"name":"demo","version":"1.0"}}}'

SID2=<paste>

## The long call. -N or curl buffers the frames; the Accept header or it never streams
## at all (a tools/call, a progressToken AND text/event-stream, or it falls back to
## plain JSON with no complaint). DjangoTestCase measured 25s cold, 4 tests, all green.
curl -N -X POST localhost:8010/mcp -H 'Content-Type: application/json' -H "MCP-Session-Id: $SID2" -H 'Accept: application/json, text/event-stream' -d '{"jsonrpc":"2.0","id":2,"method":"tools/call","params":{"name":"run_python_tests","arguments":{"classNames":["DjangoTestCase"]},"_meta":{"progressToken":"p1"}}}'

## The contrast, and the point of the demo -- same call, minus _meta. One JSON object,
## nothing for 25 seconds.
curl -N -X POST localhost:8010/mcp -H 'Content-Type: application/json' -H "MCP-Session-Id: $SID2" -H 'Accept: application/json, text/event-stream' -d '{"jsonrpc":"2.0","id":3,"method":"tools/call","params":{"name":"run_python_tests","arguments":{"classNames":["DjangoTestCase"]}}}'
